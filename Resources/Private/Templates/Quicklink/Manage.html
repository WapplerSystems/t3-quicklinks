document.addEventListener('DOMContentLoaded', function() { const activeList = document.getElementById('ws-quicklinks-active'); const availableList = document.getElementById('ws-quicklinks-available'); function saveActiveOrder() { const order = Array.from(activeList.children) .map(li => li.dataset.uid) .filter(Boolean) .join(','); console.debug(order); document.cookie = "quicklinks_order=" + order + ";path=/"; } // Initialisiere Sortable für aktive Liste (sortierbar und verschiebbar) Sortable.create(activeList, { group: 'quicklinks', animation: 150, onAdd: saveActiveOrder, onUpdate: saveActiveOrder, onRemove: saveActiveOrder, onSort: saveActiveOrder }); // Initialisiere Sortable für verfügbare Liste (nur verschiebbar) Sortable.create(availableList, { group: 'quicklinks', animation: 150, onAdd: saveActiveOrder, onRemove: saveActiveOrder }); // Beim Laden: Reihenfolge aus Cookie wiederherstellen const cookie = document.cookie.split('; ').find(row => row.startsWith('quicklinks_order=')); if (cookie) { const uids = cookie.split('=')[1].split(','); // Alle verfügbaren Items sammeln const allItems = Array.from(availableList.querySelectorAll('.ws-quicklink__item')); // Items in aktive Liste verschieben und sortieren uids.forEach(uid => { const item = allItems.find(li => li.dataset.uid === uid); if (item) { activeList.appendChild(item); } }); } // Checkbox-Logik entfernen, da Drag & Drop verwendet wird });
