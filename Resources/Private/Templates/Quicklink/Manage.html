document.addEventListener('DOMContentLoaded', function () { const wrapper = document.querySelector('.ws-quicklinks-manage'); const activeList = document.getElementById('ws-quicklinks-active'); const availableList = document.getElementById('ws-quicklinks-available'); if (!wrapper || !activeList || !availableList) { return; } const cookieName = wrapper.dataset.cookieName || 'quicklinks_order'; const saveUri = location.pathname + '?tx_wsquicklinks_save=1'; function persistOrder() { const order = Array.from(activeList.children) .map(li => li.dataset.uid) .filter(Boolean) .join(','); // Anonymous fallback: always keep the cookie in sync. const secure = location.protocol === 'https:' ? '; Secure' : ''; document.cookie = cookieName + '=' + encodeURIComponent(order) + '; path=/; max-age=31536000; SameSite=Lax' + secure; // Logged-in users additionally get it stored server-side (fe_user record). const params = new URLSearchParams(); params.append('order', order); fetch(saveUri, { method: 'POST', headers: {'X-Requested-With': 'XMLHttpRequest'}, credentials: 'same-origin', body: params }).catch(() => { /* cookie fallback already applied */ }); } Sortable.create(activeList, { group: 'quicklinks', animation: 150, onSort: persistOrder }); Sortable.create(availableList, { group: 'quicklinks', animation: 150, onSort: persistOrder }); });
